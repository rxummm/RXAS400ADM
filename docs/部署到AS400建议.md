import org.springframework.boot.web.server.WebServerFactoryCustomizer;
import org.springframework.boot.web.servlet.server.ConfigurableServletWebServerFactory;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.http.HttpStatus;

@Configuration
public class ViewControllerConfig {
    @Bean
    public WebServerFactoryCustomizer<ConfigurableServletWebServerFactory> containerCustomizer() {
        return container -> {
            // 拦截 404 错误并内部重定向到 index.html
            ErrorPage error404Page = new ErrorPage(HttpStatus.NOT_FOUND, "/index.html");
            container.addErrorPages(error404Page);
        };
    }
}

cd "$(dirname "$0")"

# 显式指定 Java 路径，防止环境变量丢失
export JAVA_HOME=/QOpenSys/QIBM/ProdData/JavaVM/jdk17/64bit  # 改为实际JDK路径
export PATH=$JAVA_HOME/bin:$PATH

APP_NAME="your-combined-app.jar"
PORT=8080

# 检查是否已经在运行
PID=$(ps -ef | grep $APP_NAME | grep -v grep | awk '{print $2}')
if [ ! -z "$PID" ]; then
    echo "警告: $APP_NAME 已经在运行中 (PID: $PID)"
    exit 1
fi

# 限制 JVM 最大物理内存为 2GB，并在启动时强制声明 UTF-8 编码
nohup java -Xms512m -Xmx2048m -Dfile.encoding=UTF-8 -jar $APP_NAME --server.port=$PORT > output.log 2>&1 &
disown

echo "服务已在后台启动，尝试访问: http://<AS400_IP>:$PORT"

APP_NAME="your-combined-app.jar"

PID=$(ps -ef | grep $APP_NAME | grep -v grep | awk '{print $2}')

if [ -z "$PID" ]; then
    echo "没有找到正在运行的 $APP_NAME 服务。"
else
    echo "正在停止进程 (PID: $PID) ..."
    kill -15 $PID
    sleep 3
    
    # 二次检查，若未安全退出则强杀
    PID_CHECK=$(ps -ef | grep $APP_NAME | grep -v grep | awk '{print $2}')
    if [ ! -z "$PID_CHECK" ]; then
        kill -9 $PID_CHECK
    fi
    echo "服务已成功停止。"
fi
